/**
 * Creates a Leva color input with per-input onChange that sets CSS
 * custom properties directly on :root, bypassing React re-renders.
 * Leva sets `transient: true` by default when onChange is provided.
 */
function tokenColor(defaultValue: string, appliedKeys: Set<string>) {
  return {
    value: defaultValue,
    onChange: (value: string, path: string) => {
      const key = path.split('.').pop() || path;
      document.documentElement.style.setProperty(key, value);
      appliedKeys.add(key);
    },
  };
}

/**
 * Creates a Leva numeric input with per-input onChange.
 * Appends 'ms' for duration tokens, 'px' for everything else.
 */
function tokenNumber(
  defaultValue: number,
  options: { min: number; max: number; step: number },
  appliedKeys: Set<string>,
) {
  return {
    value: defaultValue,
    ...options,
    onChange: (value: number, path: string) => {
      const key = path.split('.').pop() || path;
      const unit = MS_TOKENS.has(key) ? 'ms' : 'px';
      document.documentElement.style.setProperty(key, `${value}${unit}`);
      appliedKeys.add(key);
    },
  };
}

const MS_TOKENS = new Set([
  '--duration-fast',

/**
 * Creates a Leva color input with per-input onChange that sets CSS
 * custom properties directly on :root, bypassing React re-renders.
 * Leva sets `transient: true` by default when onChange is provided.
 */
function tokenColor(defaultValue: string, appliedKeys: Set<string>) {
  return {
    value: defaultValue,
    onChange: (value: string, path: string) => {
      const key = path.split('.').pop() || path;
      document.documentElement.style.setProperty(key, value);
      appliedKeys.add(key);
    },
  };
}

/**
 * Creates a Leva numeric input with per-input onChange.
 * Appends 'ms' for duration tokens, 'px' for everything else.
 */
function tokenNumber(
  defaultValue: number,
  options: { min: number; max: number; step: number },
  appliedKeys: Set<string>,
) {
  return {
    value: defaultValue,
    ...options,
    onChange: (value: number, path: string) => {
      const key = path.split('.').pop() || path;
      const unit = MS_TOKENS.has(key) ? 'ms' : 'px';
      document.documentElement.style.setProperty(key, `${value}${unit}`);
      appliedKeys.add(key);
    },
  };
}

```css
box-shadow:
  0 0 8px 0 var(--color-cga-amber-glow),    /* inner core */
  0 0 20px 2px var(--effects-phosphor-glow), /* mid bloom */
  0 0 40px 4px var(--effects-bloom-outer);   /* outer halo */
```

Text gains a subtle `text-shadow` glow. No size change — the bloom sits
entirely in box-shadow space.

### Active: Steady Energy
Background brightens to `--color-cga-amber-bright`. Glow intensifies.
A slow-crawling scanline overlay (`::after` pseudo-element) scrolls down
at 4s per cycle, simulating CRT scan refresh.

### Hover: Phosphor Warmup Flicker
On hover entry, a `phosphor-warmup` keyframe plays once (400ms). It simulates
CRT phosphor ignition: a brightness spike → overcorrection dip → secondary
excitation → settle. This runs simultaneously with the bloom transition.

### Active: Phosphor Energize Flash
On press, a `phosphor-energize` keyframe fires (150ms). A quick brightness
flash to 1.4x then holds at 1.08x while the button is held. The scanline
crawl overlay continues independently.

### Release: Power-On Impulse
On mouse-up, properties snap back with a 600ms spring curve:
`cubic-bezier(0.8, -0.5, 0.2, 1.4)`. The negative p2 creates a brief
undershoot (dips cold) and p4 > 1 overshoots past rest — a damped oscillation
that feels like phosphor decay.

### Secondary: 2-Layer Bloom

The secondary button has a softer phosphor system:

| State | Behavior |
|-------|----------|
| **Hover** | 2-layer box-shadow (inner core + mid bloom), text-shadow glow, warmup flicker |
| **Active** | Intensified 2-layer glow, energize flash (no scanline) |
| **Release** | Gentler spring `cubic-bezier(0.5, -0.2, 0.2, 1.2)` over 400ms |

### Ghost: 1-Layer Bloom

The ghost button has the most minimal phosphor treatment:

| State | Behavior |
|-------|----------|
| **Hover** | 1-layer box-shadow (inner core only), text-shadow glow, fast warmup (200ms) |
| **Active** | Intensified 1-layer glow, energize flash (no scanline) |
| **Release** | Subtle spring `cubic-bezier(0.5, -0.1, 0.2, 1.1)` over 300ms |

### Link: Text Glow Only

Link buttons receive only a `text-shadow` glow on hover — no box-shadow bloom
or spring easing. This keeps them lightweight for inline use.

Each variant maps to a different navigation context:

| Variant | Style | Use Case |
|---------|-------|----------|

| **Pills** | Contained background fill | App navigation, toolbar mode selection |

Three sizes following an 8px spacing grid:

| Size | Font | Padding (Underline) | Padding (Pills) |
|------|------|---------------------|-----------------|

## States

### Active Tab

The active tab has full opacity. In underline variant, it gains a bottom border
in the accent color. In pills variant, it fills with the accent background and
switches to secondary (dark) text.

### Inactive Tab

Inactive tabs render at 0.7 opacity (tuned for WCAG AA 4.5:1 contrast ratio).
On hover, opacity returns to 1 and a subtle `text-shadow` glow appears.

### Disabled Tab

Disabled tabs render at 0.3 opacity with `not-allowed` cursor. They are skipped
during keyboard navigation and cannot be activated.

Pass `activeTab` and `onTabChange` for full external control:

```tsx
const [activeTab, setActiveTab] = useState('schedule');
<Tabs tabs={tabs} activeTab={activeTab} onTabChange={setActiveTab} />
```

| `End` | Move to last enabled tab |
